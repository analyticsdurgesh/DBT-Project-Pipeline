# dbt adapter for Snowflake.
# Installing this also installs dbt-core and the Snowflake Python connector.
# Ready-made builds only for cryptography: its newest versions have no Intel-Mac build, and compiling it
# needs Rust. pip then takes the newest version that has a build for the learner's computer.
--only-binary=cryptography
dbt-snowflake>=1.9,<2.0
# A released dbt, never a test build: dbt-snowflake accepts pre-releases, and pip then picks one.
dbt-core>=1.10,<1.13
